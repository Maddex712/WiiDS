cmake_minimum_required(VERSION 3.20)
project(wiids_disc_host_test C CXX)
# Host build of wiids_disc + its test tool, with zlib-ng (zlib-compatible) from Dolphin's checkout.
set(ZLIB_COMPAT ON CACHE BOOL "" FORCE)
set(ZLIB_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ZLIBNG_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../.tools/src/dolphin/Externals/zlib-ng/zlib-ng zlib-ng EXCLUDE_FROM_ALL)
set(WIIDS_DISC_ZLIB_TARGET zlib)
set(WIIDS_DISC_TOOL ON)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../wiids/disc wiids_disc)
set(WIIDS_ART_ZLIB_TARGET zlib)
set(WIIDS_ART_TOOL ON)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../wiids/art wiids_art)
