cmake_minimum_required(VERSION 3.22)
project(wiids_catalog_archive C)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This target is the Android ARM64 catalog archive JNI library")
endif()
foreach(required WIIDS_ARCHIVE_SOURCE WIIDS_ARCHIVE_LIBRARY WIIDS_LZMA_SOURCE WIIDS_LZMA_LIBRARY WIIDS_CATALOG_JNI)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Supply ${required}")
  endif()
endforeach()
add_library(wiids_catalog_archive SHARED "${WIIDS_CATALOG_JNI}")
target_include_directories(wiids_catalog_archive PRIVATE
  "${WIIDS_ARCHIVE_SOURCE}/libarchive" "${WIIDS_LZMA_SOURCE}/src/liblzma/api")
target_compile_definitions(wiids_catalog_archive PRIVATE LIBARCHIVE_STATIC)
target_compile_options(wiids_catalog_archive PRIVATE -O2 -ffunction-sections -fdata-sections)
find_library(z_library z REQUIRED)
target_link_libraries(wiids_catalog_archive PRIVATE "${WIIDS_ARCHIVE_LIBRARY}" "${WIIDS_LZMA_LIBRARY}" ${z_library})
target_link_options(wiids_catalog_archive PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
