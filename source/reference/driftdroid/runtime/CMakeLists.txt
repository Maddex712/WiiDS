    # libnx gives every pthread/std::thread a 128 KiB stack; Tint's WGSL resolver and mesa's GLSL
    # compiler overflow that. __wrap___syscall_thread_create (main.cpp) raises the floor for every
    # thread in the process, including ones created inside Dawn, SDL and libstdc++.
    add_link_options(-Wl,--wrap=__syscall_thread_create)

    # a local script) at the front of .text. 43 MB of translated code otherwise
    # scatters the hot set across dozens of 1 MiB regions and thousands of pages the A57's TLBs
    # and L2 cannot hold; measured 2026-09-25: 28 -> 30 FPS stock, 40 -> 50 FPS overclocked.
    # The file must begin with KEEP (*(.crt0)): hbloader enters the NRO at offset 0.
    option(MKW_SWITCH_SECTION_ORDERING "Order hot functions first in the Switch NRO" ON)
    set(MKW_SWITCH_SECTION_ORDER_FILE "${CMAKE_CURRENT_SOURCE_DIR}/tools/switch_section_order.ld")
    if(MKW_SWITCH_SECTION_ORDERING AND EXISTS "${MKW_SWITCH_SECTION_ORDER_FILE}")
        add_link_options("-Wl,--section-ordering-file=${MKW_SWITCH_SECTION_ORDER_FILE}")
    endif()
endif()

# Profile-guided optimization for the Switch build. Two-pass:
#   cmake -DMKW_PGO=generate ...  -> instrumented build; play, then the runtime calls
#                                   __gcov_dump() and the .gcda files land under
#                                   sdmc:/switch/WiiCompiled/Cache/gcov (GCOV_PREFIX).
#   copy that tree back over the build dir, then
#   cmake -DMKW_PGO=use ...      -> optimized build using the measured hot paths.
set(MKW_PGO "" CACHE STRING "PGO pass (Switch and Android): empty, generate or use")
# Android (clang): the recording build writes files/WiiCompiled/pgo/game.profraw on the device
# (runtime/src/android_pgo.cpp). Pull it, run `llvm-profdata merge -o android.profdata`, and pass
# the result as MKW_PGO_PROFILE to the "use" build.
set(MKW_PGO_PROFILE "" CACHE FILEPATH "Android PGO: merged .profdata for -DMKW_PGO=use")
if(ANDROID AND MKW_PGO STREQUAL "generate")
    add_compile_options(-fprofile-generate)
    add_link_options(-fprofile-generate)
    list(APPEND MKW_PROJECT_COMPILE_DEFINITIONS MKW_PGO_GENERATE=1)
elseif(ANDROID AND MKW_PGO STREQUAL "use")
    if(NOT EXISTS "${MKW_PGO_PROFILE}")
        message(FATAL_ERROR "MKW_PGO=use on Android needs MKW_PGO_PROFILE=<merged .profdata>")
    endif()
    # A source edited after the recording only loses its own profile instead of failing the build.
    add_compile_options("-fprofile-use=${MKW_PGO_PROFILE}" -Wno-profile-instr-unprofiled
                        -Wno-profile-instr-out-of-date -Wno-backend-plugin)
elseif(MKW_PGO STREQUAL "generate")
    # single, not atomic: the A57 is ARMv8.0 with no LSE, so every atomic counter bump would be an
    # exclusive-monitor retry loop and the training run would crawl. Translated code runs on one
    # thread; the few racy counters elsewhere are what -fprofile-correction absorbs below.
    # -fno-profile-values: devkitA64's prebuilt libgcov is built for -mtp=el0, so its indirect-call
    # profiler reads TLS through TPIDR_EL0, which Horizon resets. The first indirect call during
    # static init wrote through a garbage pointer and the recording build hung on a black screen
    # before main. Arc (branch/block) counts, the part layout and hot/cold splitting need, stay on.
    add_compile_options(-fprofile-generate -fprofile-update=single -fno-profile-values)
    add_link_options(-fprofile-generate -fprofile-update=single -fno-profile-values)
    list(APPEND MKW_PROJECT_COMPILE_DEFINITIONS MKW_PGO_GENERATE=1)
elseif(MKW_PGO STREQUAL "use")
    # -fprofile-correction: counters are written from several threads; missing or
    # inconsistent counts must not fail the build.
    # -fprofile-use also switches on -O3-class loop transforms; keep them off (code growth is a
    # measured loss on this i-cache-bound binary) and let the profile drive layout instead.
    # No -freorder-blocks-and-partition: devkitA64's GCC emits jump-table entries that subtract
    # labels across the hot and .text.unlikely sections, which the assembler rejects ("invalid
    # operands ... for `-'"). -fno-reorder-functions keeps every function in `.text.<name>`
    # (instead of .text.hot./.text.unlikely.) so MKW_SWITCH_SECTION_ORDERING's measured race order
    # still applies to the profiled build.
    # -fno-profile-values matches the generate pass (see above), which records no value counters.
    # -Wno-error=coverage-mismatch: a source edited after the recording build only loses its own
    # profile (GCC warns and falls back) instead of failing the whole build.
    add_compile_options(-fprofile-use -fprofile-correction -Wno-missing-profile -fno-profile-values
                        -Wno-error=coverage-mismatch -fno-reorder-functions
                        -fno-unroll-loops -fno-peel-loops -fno-unswitch-loops -fno-split-loops
                        -fno-tree-loop-distribution -fno-loop-interchange -fno-loop-unroll-and-jam
                        -fno-predictive-commoning -fno-version-loops-for-strides -fno-ipa-cp-clone
                        -fno-tracer -fvect-cost-model=very-cheap)
    add_link_options(-fprofile-use -fprofile-correction)
elseif(NOT MKW_PGO STREQUAL "")
    message(FATAL_ERROR "MKW_PGO must be empty, generate or use")
endif()

# Wii ES signatures use the standard binary-field sect233r1 curve. Crypto++
# supplies that curve, SHA-1, ECDSA and an OS-seeded CSPRNG. The complete 8.9.0
# source distribution is vendored. Assembly and algorithm-specific SIMD units
# are disabled to keep the library portable and independent of the runtime's
# x86-64-v3 policy; the static linker retains only the algorithms we reference.
#
# Crypto++ is by far the most expensive vendored library to compile (~150
# translation units), and under the shipped fixed toolchain it is bit-for-bit
# identical for every user, so the prebuilt package harvests its archive and
# this block is skipped entirely in that mode (see cmake/NativePrebuilt.cmake).
if(NOT MKW_NATIVE_PREBUILT_DIR)
    file(GLOB MKW_CRYPTOPP_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_LIST_DIR}/third_party/cryptopp/*.cpp")
    list(FILTER MKW_CRYPTOPP_SOURCES EXCLUDE REGEX
        "/(adhoc|bench[123]|datatest|dlltest|fipsalgt|fipstest|pch|regtest[1234]|simple|test|validat[0-9]+)\\.cpp$")
    list(FILTER MKW_CRYPTOPP_SOURCES EXCLUDE REGEX
        "(_avx|_ppc|_simd|_sse)\\.cpp$")
    add_library(mkw_cryptopp STATIC ${MKW_CRYPTOPP_SOURCES})
    add_library(mkw::cryptopp ALIAS mkw_cryptopp)
    target_include_directories(mkw_cryptopp SYSTEM PUBLIC third_party)
    target_compile_features(mkw_cryptopp PUBLIC cxx_std_17)
    target_compile_definitions(mkw_cryptopp PUBLIC ${MKW_CRYPTOPP_INTERFACE_DEFINITIONS})
    target_compile_options(mkw_cryptopp PRIVATE -w)
    set_target_properties(mkw_cryptopp PROPERTIES UNITY_BUILD OFF)
    if(MKW_TARGET_ANDROID)
        # cpu.cpp unconditionally does `#if defined(__ANDROID__) # include "cpu-features.h"
        # #endif` and calls android_getCpuFamily()/android_getCpuFeatures() to gate hardware-

        # every NDK at $ANDROID_NDK_ROOT/sources/android/cpufeatures, but modern NDKs (confirmed:
        # r27c, the one this project uses) dropped it - android_getCpuFeatures() was deprecated in
        # favor of getauxval(AT_HWCAP). This project already disables Crypto++'s ASM/SIMD code
        # entirely via CRYPTOPP_DISABLE_ASM (see the comment above this block - "portable and
        # independent of the runtime's x86-64-v3 policy"), so the feature-detection RESULT these

        # compile. Same stub-header trick this file already uses for DXProgrammableCapture.h on

        # and the semantically correct answer given ASM is off project-wide.
        set(MKW_ANDROID_CPU_FEATURES_STUB_DIR "${CMAKE_BINARY_DIR}/android_cpu_features_stub")

// Stub for the NDK cpu-features.h this project's vendored Crypto++ still references (removed

// runtime/CMakeLists.txt for the full explanation.
#include <stdint.h>
typedef enum { ANDROID_CPU_FAMILY_UNKNOWN = 0, ANDROID_CPU_FAMILY_ARM = 1, ANDROID_CPU_FAMILY_ARM64 = 2 } AndroidCpuFamily;
enum {

    endif()
    # Switch: aurora-main (SDL3 windowing + Dawn/WebGPU) has no Switch backend at all and cannot
    # get one - Dawn only targets Vulkan/D3D/Metal/GL, none of which exist on stock Switch
    # homebrew (graphics go through devkitPro's own deko3d instead). This add_subdirectory() is
    # still attempted for Switch rather than skipped, deliberately: Phase 1's own goal is to prove
    # this project's CMake logic gets as far as it can and fails exactly at the SDL3/Dawn boundary,
    # not to paper over that boundary early. Expect this configure (or the following build) to
    # fail somewhere in aurora-main/Dawn for MKW_TARGET_SWITCH - that failure is Phase 1's real
    # deliverable, and Phase 3 replaces aurora_core's GPU/window layer wholesale for this target
    # rather than trying to make Dawn itself work here.
    set(DAWN_ENABLE_D3D11 OFF CACHE BOOL "" FORCE)
    if(WIN32)
        set(DAWN_ENABLE_D3D12 ON CACHE BOOL "" FORCE)
        set(TINT_BUILD_HLSL_WRITER ON CACHE BOOL "" FORCE)
        set(DAWN_USE_WINDOWS_UI OFF CACHE BOOL "" FORCE)
    else()
        # Non-Windows (Linux/Android): mirrors aurora-main's own
        # _aurora_dawn_set_platform_backends() choice for this platform - Vulkan only, no
        # D3D/HLSL. Kept in sync here because this project's own CMake FORCEs these cache
        # variables before aurora-main's add_subdirectory() runs, which pre-empts aurora's
        # auto-detection (CACHE ... INTERNAL "" without FORCE never overrides an existing value).
        set(DAWN_ENABLE_D3D12 OFF CACHE BOOL "" FORCE)
        set(TINT_BUILD_HLSL_WRITER OFF CACHE BOOL "" FORCE)
    endif()
    set(DAWN_ENABLE_VULKAN ON CACHE BOOL "" FORCE)
    if(MKW_TARGET_ANDROID)
        # Android is the one target where Vulkan is not guaranteed present/functional on every
        # device (older or budget SoCs, or a buggy vendor driver) - unlike desktop Linux, where
        # this project treats Vulkan as the only backend. Hence the graphics
        # decision (Vulkan primary, OpenGL ES fallback): aurora's own platform-backend function defaults OpenGLES OFF for Android
        # (it falls into the generic "else" branch there), so it must be forced ON here the same
        # way the other DAWN_ENABLE_* flags already are.
        set(DAWN_ENABLE_OPENGLES ON CACHE BOOL "" FORCE)

        # AuroraDawnProvider.cmake's "auto" resolution only recognizes prebuilt Dawn PACKAGES for
        # windows/linux-x86_64/linux-aarch64/macos - not Android - so it falls through to "vendor"
        # (build Dawn from source via FetchContent), whose pinned URL
        # (github.com/google/dawn/archive/refs/tags/${AURORA_DAWN_VERSION}.tar.gz) 404s: that git
        # tag has been pruned from the real google/dawn repo (confirmed directly). The dawn-build release at the SAME
        # AURORA_DAWN_VERSION this project already pins everywhere else DOES publish
        # dawn-android-aarch64.tar.gz (confirmed via the GitHub releases API) - forcing "package"
        # here uses that prebuilt directly and skips the broken vendor path entirely. KartPad (a
        # sibling static-recompilation port of this project for Apple platforms) does the exact
        # same thing for its own targets: AURORA_DAWN_PROVIDER=package at this same pinned
        # version, never vendor.
        set(AURORA_DAWN_PROVIDER "package" CACHE STRING "" FORCE)

        # The dawn-android-aarch64 release asset's exported lib/cmake/Dawn/DawnTargets.cmake bakes
        # an ABSOLUTE, non-relocatable path to liblog.so from the machine that built the package
        # (something like /usr/local/lib/android/sdk/ndk/<their-ndk-version>/.../liblog.so) into
        # INTERFACE_LINK_LIBRARIES, instead of a portable "log"/"-llog" reference - a packaging bug
        # in that specific release (verified directly: the path doesn't exist on this machine and
        # points at a different NDK version than the one actually configured here), not anything under this project's control to fix at
        # the source. Pre-fetch the SAME package AuroraDawnProvider.cmake's "package" branch will
        # ask for below (identical FetchContent NAME/URL, so its own later
        # FetchContent_Declare/MakeAvailable calls become a documented no-op reuse of what's
        # already populated here) and rewrite the one bad path to point at THIS machine's real NDK
        # liblog.so instead, derived from CMAKE_TOOLCHAIN_FILE rather than assumed.
        # AURORA_DAWN_VERSION itself isn't defined yet in this scope - it's only set once
        # aurora-main's own CMakeLists.txt runs, which is the add_subdirectory() call below this
        # whole block. Read the same pin directly out of that file instead of duplicating the
        # literal version string here (which would silently drift out of sync if aurora-main's own
        # pin is ever bumped).
        file(STRINGS "${MKW_AURORA_DIR}/CMakeLists.txt" MKW_AURORA_DAWN_VERSION_LINE
            REGEX "set\\(AURORA_DAWN_VERSION ")
        string(REGEX MATCH "\"([^\"]+)\"" MKW_AURORA_DAWN_VERSION_MATCH "${MKW_AURORA_DAWN_VERSION_LINE}")
        if(NOT CMAKE_MATCH_1)
            message(FATAL_ERROR "Could not read AURORA_DAWN_VERSION from ${MKW_AURORA_DIR}/CMakeLists.txt")
        endif()
        set(AURORA_DAWN_VERSION "${CMAKE_MATCH_1}")

        # The toolchain selects the host's NDK sysroot (Windows, macOS, or Linux).
        # Reconstructing it with a linux-x86_64 directory breaks Windows builds.
        set(MKW_ANDROID_LOG_LIB
            "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-android/28/liblog.so")
        if(NOT EXISTS "${MKW_ANDROID_LOG_LIB}")
            message(FATAL_ERROR "Could not derive this machine's liblog.so from CMAKE_SYSROOT "
                "(looked for ${MKW_ANDROID_LOG_LIB}) - the Dawn Android package patch below needs it.")
        endif()

        include(FetchContent)
        FetchContent_Declare(dawn_prebuilt
            URL "https://github.com/encounter/dawn-build/releases/download/${AURORA_DAWN_VERSION}/dawn-android-aarch64.tar.gz"
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE
        )
        FetchContent_MakeAvailable(dawn_prebuilt)
        set(MKW_DAWN_TARGETS_FILE "${dawn_prebuilt_SOURCE_DIR}/lib/cmake/Dawn/DawnTargets.cmake")
        if(EXISTS "${MKW_DAWN_TARGETS_FILE}")
            file(READ "${MKW_DAWN_TARGETS_FILE}" MKW_DAWN_TARGETS_CONTENT)
            string(REGEX REPLACE
                "/[^;\"]*/ndk/[^;\"]*/toolchains/llvm/prebuilt/linux-x86_64/sysroot/usr/lib/aarch64-linux-android/28/liblog\\.so"
                "${MKW_ANDROID_LOG_LIB}"
                MKW_DAWN_TARGETS_CONTENT_PATCHED "${MKW_DAWN_TARGETS_CONTENT}")
            if(NOT MKW_DAWN_TARGETS_CONTENT_PATCHED STREQUAL MKW_DAWN_TARGETS_CONTENT)
                file(WRITE "${MKW_DAWN_TARGETS_FILE}" "${MKW_DAWN_TARGETS_CONTENT_PATCHED}")
                message(STATUS "Patched Dawn's Android package: replaced its non-portable liblog.so path with this machine's real one")
            endif()
        endif()

        # Custom Snapdragon Vulkan drivers (Turnip, newer Qualcomm builds) through libadrenotools,
        # BSD-2-Clause, the loader emulators use. Only useful with a Dawn that carries the
        # SetVulkanLibraryOverride patch (point FETCHCONTENT_SOURCE_DIR_DAWN_PREBUILT at one); with
        # the stock package the runtime detects its absence and uses the system driver.
        option(MKW_ANDROID_CUSTOM_GPU_DRIVERS "Load user-installed Vulkan drivers via libadrenotools" ON)
        if(MKW_ANDROID_CUSTOM_GPU_DRIVERS)
            FetchContent_Declare(adrenotools
                GIT_REPOSITORY https://github.com/bylaws/libadrenotools.git
                GIT_TAG 8fae8ce254dfc1344527e05301e43f37dea2df80
                GIT_SUBMODULES_RECURSE TRUE
            )
            FetchContent_MakeAvailable(adrenotools)
        endif()
    else()
        set(DAWN_ENABLE_OPENGLES OFF CACHE BOOL "" FORCE)
    endif()
    set(DAWN_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
    set(DAWN_BUILD_TESTS OFF CACHE BOOL "" FORCE)

    # Provide a tiny stub for DXProgrammableCapture when the SDK/PIX headers are
    # missing (common on MinGW). Dawn only includes the header; no symbols are
    # referenced when PIX isn't present.
    set(MKW_DX_STUB_DIR "${CMAKE_BINARY_DIR}/aurora_dx_stubs")
    if(NOT EXISTS "${MKW_DX_STUB_DIR}/DXProgrammableCapture.h")
        file(MAKE_DIRECTORY ${MKW_DX_STUB_DIR})
        file(WRITE "${MKW_DX_STUB_DIR}/DXProgrammableCapture.h"
"#pragma once\n// Stubbed PIX capture header for Dawn; no functionality when PIX is absent.\n")
    endif()
    # Deliberately NOT injected project-wide. Only a from-source Dawn build ever
    # includes DXProgrammableCapture.h, and this tree consumes Dawn as a prebuilt
    # package (AURORA_DAWN_PROVIDER=package), so the old
    # include_directories()/add_compile_options(-I...) pair only added a dead
    # -I to every one of the ~500 third-party and runtime compile lines. The
    # aurora_* targets that do want it get it from the
    # target_include_directories() call further down; a from-source Dawn build
    # should add it to the Dawn targets the same way.

    # Keep SDL3 lean when aurora is enabled.
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)

    # MinGW/WSL can choke on SDL precompiled headers; disable PCH for aurora/SDL.
    set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
    # Isolate aurora build products to a subdirectory under the main build dir.
    add_subdirectory(${MKW_AURORA_DIR} ${CMAKE_BINARY_DIR}/aurora-build EXCLUDE_FROM_ALL)
    set(CMAKE_DISABLE_PRECOMPILE_HEADERS OFF)

    # Keep upstream Aurora split: several sources intentionally use file-local
    # names that collide under this repo's aggressive unity build settings.
    set(AURORA_TARGETS aurora_core aurora_gx aurora_pad aurora_si aurora_mtx aurora_vi)
    foreach(t ${AURORA_TARGETS})
        if(TARGET ${t})
            set_target_properties(${t} PROPERTIES UNITY_BUILD OFF)
            target_compile_options(${t} PRIVATE -O3 -ffast-math -w -pipe)
            target_include_directories(${t} PRIVATE ${MKW_DX_STUB_DIR})
            # Aurora's own sources include Windows headers and call std::min/max;
            # they relied on the old project-wide NOMINMAX that no longer leaks
            # into this subtree, so the define is applied per target here.
            target_compile_definitions(${t} PRIVATE NOMINMAX)
        endif()
    endforeach()

    # Frame interpolation filters guest NaN/Inf matrices out of a std::sort
    # comparator, which makes its std::isfinite guards load-bearing for memory
    # safety, not just match quality. -ffinite-math-only (part of -ffast-math
    # above) folds those guards to constants and poisons NaN float arguments
    # via nofpclass, so this one file opts back out of the finite-math
    # assumption; the file #errors if built without this. Source-scoped
    # COMPILE_OPTIONS append after the target options, so this overrides the
    # target's -ffast-math for exactly this TU.
    if(TARGET aurora_gx)
        set_source_files_properties("${MKW_AURORA_DIR}/lib/gx/frame_interpolation.cpp"
            TARGET_DIRECTORY aurora_gx
            PROPERTIES COMPILE_OPTIONS "-fno-finite-math-only")
    endif()

    if(MKW_NATIVE_PREBUILT_EXPORT_DIR)
        include("${CMAKE_CURRENT_LIST_DIR}/cmake/NativePrebuiltExport.cmake")
    endif()
endif()

# Everything from here down is this project's own code, so this is where the
# project definitions enter the directory scope. A directory-scoped definition
# seeds the COMPILE_DEFINITIONS of targets as they are created, which means:
#   * aurora-main and its third-party tree were added above -> unaffected;
#   * mkw_runtime_common, the translated shard libraries and the products in
#     cmake/PublicProducts.cmake are all created below (that file is include()d,
#     not add_subdirectory()d, so it shares this directory scope) -> they pick
#     the definitions up exactly as before.
if(MKW_PROJECT_COMPILE_DEFINITIONS)
    add_compile_definitions(${MKW_PROJECT_COMPILE_DEFINITIONS})
endif()

# Runtime sources. CONFIGURE_DEPENDS costs one glob re-check per build (measured
# at ~55 ms across the 100+ globs this build already re-verifies for SDL), and it
# buys correctness: without it, a newly added src/*.cpp that is not referenced by
# a registration file is silently never compiled and never errors. The stale-glob
# failure mode is worth far more than the milliseconds.
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "src/*.cpp")
set(MKW_BASE_PRODUCT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/src/product/base_product.cpp")
set(MKW_RETRO_REWIND_PRODUCT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/src/product/retro_rewind_product.cpp")
# The host ISA guard is the one translation unit that must not receive the
# x86-64-v3 flags, so it gets its own object library instead of riding along in
# mkw_runtime_common. See cmake/PublicProducts.cmake and the file's own header.
set(MKW_CPU_BASELINE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/src/host_cpu_baseline.cpp")
# Android-only combined-library provider (see cmake/PublicProducts.cmake's MKW_TARGET_ANDROID
# branch) - like the two product sources above, it defines RuntimeProduct::Active() and must
# stay out of mkw_runtime_common's shared object graph or every product duplicate-symbols it.
set(MKW_COMBINED_PRODUCT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/src/product/combined_product.cpp")
list(REMOVE_ITEM SOURCES ${MKW_BASE_PRODUCT_SOURCE} ${MKW_RETRO_REWIND_PRODUCT_SOURCE}
     ${MKW_CPU_BASELINE_SOURCE} ${MKW_COMBINED_PRODUCT_SOURCE})

    "${CMAKE_CURRENT_LIST_DIR}/../generated/build_shards/shards.cmake"

include("${MKW_TRANSLATED_SHARD_MANIFEST}")

message(STATUS
    "Translator graph: ${MKW_SHARED_BASE_FUNCTION_COUNT}/${MKW_BASE_FUNCTION_COUNT} base functions shared; "
    "${MKW_PROFILE_SENSITIVE_CALLER_COUNT} profile-sensitive callers; "
    "${MKW_RETRO_REWIND_FUNCTION_COUNT} Retro Rewind functions")
